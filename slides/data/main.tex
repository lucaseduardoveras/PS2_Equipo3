\begin{frame}{Fuente de datos}
\small
\begin{itemize}
    \item Los datos provienen de la \textit{Medición de Pobreza Monetaria y Desigualdad 2018}, elaborada por el \textit{Departamento Administrativo Nacional de Estadística (DANE)} en el marco de la misión \textit{Empalme de las Series de Empleo, Pobreza y Desigualdad (MESE)}.
    \item Esta iniciativa busca unificar y armonizar las series de empleo, pobreza y desigualdad en Colombia, fortaleciendo la consistencia de los indicadores sociales.
    \item La fuente contiene información detallada sobre ingresos, educación, ocupación, vivienda y composición familiar, tanto a nivel individual como de hogar.
    \item Los datos se obtuvieron de la plataforma \textit{Kaggle}, donde se desarrolla una competencia para predecir la condición de pobreza.
\end{itemize}
\end{frame}

%-----------------------------------------------
\begin{frame}{Procesamiento y limpieza de datos}
\small
\begin{itemize}
    \item Se integraron las bases de hogares y personas mediante el identificador único \texttt{id}.
    \item Se excluyeron las variables con más del 40\% de valores faltantes, evitando imputaciones poco confiables.
    \item La selección final se basó en criterios teóricos y empíricos sobre los determinantes de la pobreza.
    \item Las variables de ingreso del conjunto de entrenamiento no se incluyeron, ya que no estaban disponibles en el conjunto de prueba.
\end{itemize}

\medskip
\textbf{Tamaño muestral:} \\
164.959 hogares en el conjunto de entrenamiento y 66.168 en el de prueba (40\% del primero).
\end{frame}

%-----------------------------------------------
\begin{frame}{Variables derivadas del nivel individual}
\small
\begin{itemize}
    \item Las variables provenientes de la base de personas corresponden a:
    \begin{itemize}
        \item Sumas o proporciones de individuos que cumplen determinada condición (ocupados, inactivos, con educación superior, etc.).
        \item Características del jefe o jefa del hogar (sexo, edad, ocupación, condición laboral).
    \end{itemize}
    \item Estas medidas permiten aproximar la \textbf{estructura demográfica}, el \textbf{capital humano} y la \textbf{inserción laboral} del hogar.
\end{itemize}
\end{frame}

%-----------------------------------------------
\begin{frame}{Variables del mercado laboral}
\small
\begin{itemize}
    \item Se construyeron las tasas de \textbf{ocupación} e \textbf{inactividad}, definidas como:
    \[
    t_{ocu} = \frac{\text{Ocupados}}{\text{Personas en edad de trabajar}}, \quad
    t_{inac} = \frac{\text{Inactivos}}{\text{Personas en edad de trabajar}}
    \]
    \item Hipótesis: mayor participación laboral $\Rightarrow$ menor probabilidad de pobreza.
    \item Otras variables laborales incluidas:
    \begin{itemize}
        \item Horas totales trabajadas por los miembros del hogar.
        \item Número de trabajadores por cuenta propia.
        \item Dicotómica: jefe(a) de hogar cuenta propia.
    \end{itemize}
    \item Estas variables reflejan la \textbf{resiliencia económica} del hogar ante choques del mercado laboral.
\end{itemize}
\end{frame}

%-----------------------------------------------
\begin{frame}{Características socioeconómicas y demográficas}
\small
\begin{itemize}
    \item Variables demográficas y de género:
    \begin{itemize}
        \item Número de mujeres.
        \item Jefe(a) de hogar mujer (binaria).
        \item Tasa de dependencia:
        \[
        t_{dep} = \frac{\text{Población <15 o >65}}{\text{Población en edad de trabajar}}
        \]
    \end{itemize}
    \item Variables educativas:
    \begin{itemize}
        \item Proporción de personas con educación superior.
        \item Nivel educativo máximo alcanzado en el hogar.
    \end{itemize}
    \item Estas medidas capturan el \textbf{capital humano} y su efecto sobre los ingresos y la movilidad social.
\end{itemize}
\end{frame}

%-----------------------------------------------
\begin{frame}{Condiciones habitacionales y geográficas}
\small
\begin{itemize}
    \item Variables de vivienda:
    \begin{itemize}
        \item Número de habitaciones.
        \item Tipo de vivienda (propia, arrendada, usufructo, ocupada sin título).
    \end{itemize}
    \item Variable demográfica de control:
    \begin{itemize}
        \item Número total de personas en el hogar.
    \end{itemize}
    \item Variable geográfica:
    \begin{itemize}
        \item Dominio (área metropolitana o rural).
    \end{itemize}
    \item Estas variables complementan las dimensiones laboral y educativa, reflejando el \textbf{bienestar material} y las \textbf{condiciones estructurales del hogar}.
\end{itemize}
\end{frame}

\begin{frame}{Conclusiones descriptivas}
\small
\begin{itemize}
    \item Los hogares pobres presentan mayor número de integrantes, más menores y mayores tasas de inactividad.
    \item Exhiben menor nivel educativo y mayor proporción de trabajo por cuenta propia.
    \item Los hogares no pobres muestran una inserción laboral más estable y mejores condiciones de vivienda.
    \item Estas diferencias confirman la coherencia teórica de las variables incluidas en el modelo predictivo.
\end{itemize}
\end{frame}
